import json
import pytest
from application.models import db, Product

def test_change_password_success(client, auth_header_owner):
    """Test http://127.0.0.1:5000/api/change-password Put - Success"""
    data = {
        "current_password": "password",
        "new_password": "new_password123",
        "confirm_password": "new_password123"
    }
    response = client.put('/api/change-password', 
                          data=json.dumps(data), 
                          headers=auth_header_owner,
                          content_type='application/json')
    assert response.status_code == 200
    assert response.json['message'] == "Password changed successfully"

def test_change_password_mismatch(client, auth_header_owner):
    """Test http://127.0.0.1:5000/api/change-password Put - Mismatch"""
    data = {
        "current_password": "password",
        "new_password": "new_password123",
        "confirm_password": "different_password"
    }
    response = client.put('/api/change-password', 
                          data=json.dumps(data), 
                          headers=auth_header_owner,
                          content_type='application/json')
    assert response.status_code == 400
    assert response.json['message'] == "New password and confirm password do not match"
